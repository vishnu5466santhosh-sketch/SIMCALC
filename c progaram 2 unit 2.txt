
#include <stdio.h>
#include <string.h>

#define MAX 50
#define SIZE 50

int id[MAX], availability[MAX];
char name[MAX][SIZE];
char author[MAX][SIZE];
char category[MAX][SIZE];
float price[MAX];

int count = 0;

int searchID(int x)
{
    int i;
    for(i = 0; i < count; i++)
    {
        if(id[i] == x)
            return i;
    }
    return -1;
}

void addBook()
{
    int x;

    if(count >= MAX)
    {
        printf("Library is Full!\n");
        return;
    }

    printf("Enter Book ID: ");
    scanf("%d", &x);

    if(searchID(x) != -1)
    {
        printf("Duplicate ID!\n");
        return;
    }

    id[count] = x;

    printf("Enter Book Name: ");
    scanf(" %49[^\n]", name[count]);

    printf("Enter Author Name: ");
    scanf(" %49[^\n]", author[count]);

    printf("Enter Category: ");
    scanf(" %49[^\n]", category[count]);

    printf("Enter Price: ");
    scanf("%f", &price[count]);

    availability[count] = 1;
    count++;

    printf("Book Added Successfully!\n");
}

void displayBooks()
{
    int i;

    if(count == 0)
    {
        printf("No Books Available!\n");
        return;
    }

    for(i = 0; i < count; i++)
    {
        printf("\nID: %d", id[i]);
        printf("\nName: %s", name[i]);
        printf("\nAuthor: %s", author[i]);
        printf("\nCategory: %s", category[i]);
        printf("\nPrice: %.2f", price[i]);

        if(availability[i] == 1)
            printf("\nStatus: Available\n");
        else
            printf("\nStatus: Issued\n");
    }
}

void searchById()
{
    int x, pos;

    printf("Enter Book ID: ");
    scanf("%d", &x);

    pos = searchID(x);

    if(pos == -1)
        printf("Book Not Found!\n");
    else
    {
        printf("Book Name: %s\n", name[pos]);
        printf("Author: %s\n", author[pos]);
        printf("Price: %.2f\n", price[pos]);
    }
}

void searchByName()
{
    char x[SIZE];
    int i, found = 0;

    printf("Enter Book Name: ");
    scanf(" %49[^\n]", x);

    for(i = 0; i < count; i++)
    {
        if(strcmp(name[i], x) == 0)
        {
            printf("Book Found!\n");
            printf("ID: %d\n", id[i]);
            printf("Author: %s\n", author[i]);
            found = 1;
        }
    }

    if(found == 0)
        printf("Book Not Found!\n");
}

void issueBook()
{
    int x, pos;
    static int issueCount = 0;

    printf("Enter Book ID to Issue: ");
    scanf("%d", &x);

    pos = searchID(x);

    if(pos == -1)
        printf("Book Not Found!\n");
    else if(availability[pos] == 0)
        printf("Book Already Issued!\n");
    else
    {
        availability[pos] = 0;
        issueCount++;
        printf("Book Issued Successfully!\n");
        printf("Total Issue Operations: %d\n", issueCount);
    }
}

void returnBook()
{
    int x, pos;

    printf("Enter Book ID to Return: ");
    scanf("%d", &x);

    pos = searchID(x);

    if(pos == -1)
        printf("Book Not Found!\n");
    else if(availability[pos] == 1)
        printf("Book Already Available!\n");
    else
    {
        availability[pos] = 1;
        printf("Book Returned Successfully!\n");
    }
}

void updateBook()
{
    int x, pos;

    printf("Enter Book ID to Update: ");
    scanf("%d", &x);

    pos = searchID(x);

    if(pos == -1)
    {
        printf("Book Not Found!\n");
        return;
    }

    printf("Enter New Book Name: ");
    scanf(" %49[^\n]", name[pos]);

    printf("Enter New Author Name: ");
    scanf(" %49[^\n]", author[pos]);

    printf("Enter New Category: ");
    scanf(" %49[^\n]", category[pos]);

    printf("Enter New Price: ");
    scanf("%f", &price[pos]);

    printf("Book Updated Successfully!\n");
}

void countBooks()
{
    printf("Total Books: %d\n", count);
}

void displayAvailable()
{
    int i, found = 0;

    for(i = 0; i < count; i++)
    {
        if(availability[i] == 1)
        {
            printf("\nID: %d", id[i]);
            printf("\nName: %s\n", name[i]);
            found = 1;
        }
    }

    if(found == 0)
        printf("No Available Books!\n");
}

int main()
{
    int choice;

    do
    {
        printf("\n===== LIBRARY MANAGEMENT SYSTEM =====\n");
        printf("1. Add New Book\n");
        printf("2. Display All Books\n");
        printf("3. Search Book by ID\n");
        printf("4. Search Book by Name\n");
        printf("5. Issue Book\n");
        printf("6. Return Book\n");
        printf("7. Update Book Details\n");
        printf("8. Count Total Books\n");
        printf("9. Display Available Books\n");
        printf("10. Exit\n");

        printf("\nEnter Your Choice: ");

        if(scanf("%d", &choice) != 1)
        {
            printf("Invalid Input!\n");
            return 0;
        }

        switch(choice)
        {
            case 1: addBook(); break;
            case 2: displayBooks(); break;
            case 3: searchById(); break;
            case 4: searchByName(); break;
            case 5: issueBook(); break;
            case 6: returnBook(); break;
            case 7: updateBook(); break;
            case 8: countBooks(); break;
            case 9: displayAvailable(); break;
            case 10: printf("Thank You!\n"); break;
            default: printf("Invalid Choice!\n");
        }

    } while(choice != 10);

    return 0;
}
